#include <stdio.h>

void lerComandos(char comandos[], int tamanho) {
    for (int i = 0; i < tamanho; i++) {
        scanf(" %c", &comandos[i]);
    }
}

void processarComandos(char comandos[], int tamanho, int matriz[4][4]) {
    int x = 0, y = 0;
    matriz[y][x]++;

    for (int i = 0; i < tamanho; i++) {
        if      (comandos[i] == 'c' && y > 0) y--;
        else if (comandos[i] == 'b' && y < 3) y++;
        else if (comandos[i] == 'e' && x > 0) x--;
        else if (comandos[i] == 'd' && x < 3) x++;
        matriz[y][x]++;
    }
}

void encontrarMaisVisitado(int matriz[4][4], int resultado[2]) {
    int visitas = 0;
    resultado[0] = 0; // posx
    resultado[1] = 0; // posy

    for (int i = 0; i < 4; i++) {
        for (int j = 0; j < 4; j++) {
            if (matriz[i][j] > visitas) {
                visitas = matriz[i][j];
                resultado[1] = i; // posy
                resultado[0] = j; // posx
            }
        }
    }
}

int main() {
    char comandos[20];
    int matriz[4][4] = {0};
    int resultado[2] = {0}; // [0]=posx, [1]=posy

    lerComandos(comandos, 20);
    processarComandos(comandos, 20, matriz);
    encontrarMaisVisitado(matriz, resultado);

    printf("Coordenada X:%d, Y:%d\n", resultado[0], resultado[1]);
    return 0;
}
